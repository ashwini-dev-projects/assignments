6. answer the question for multiple inheritance
	
class A:
    def fun1(self):
        print("A")
 
class B(A):
    def fun1(self):
        print('B')
 
class C(A):
    def fun1(self):
        print("C")
 
class D(B, C):
    def fun1(self):
        super().fun1()  
 
obj = D()
obj.fun1()

def frequent_data(func):
    def wrapper(*args, **kwargs):
        data = func(*args, **kwargs)
        result = { i:data.count(i)  for i in data if data.count(i) > 1}
        # for i in data if data.count(i) > 1
        return result
    
    return wrapper

@frequent_data
def check_data(arr):  # frequent_data(check_data)
    return arr
my_list = [1, 2, 3, 4, 5, 2, 3, 6, 7, 3, 8, 2]
res = check_data(my_list)
print(res)
